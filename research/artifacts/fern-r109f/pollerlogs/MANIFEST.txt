Reproduce section 7.4 offline:
  python3 research/fern_r109f_poller_occupancy.py \
      --glob 'research/artifacts/fern-r109f/pollerlogs/*.log'

  submitted = the poller won a free slot and fired a submission
  expired   = --max-wait elapsed with the slot still busy
  watching  = the poller was still running, or was terminated,
              while the slot was occupied
